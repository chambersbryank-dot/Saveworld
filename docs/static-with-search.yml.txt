name: Deploy to GitHub Pages

on:
  push:
    branches: ["main"]
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

jobs:
  deploy:
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4

      # Search build (Oct 12): Pagefind keyword index + semantic vector index, built together
      # from the same HTML so both layers always ship in sync. See NOTES.md "Search Upgrade - Oct 12".
      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm

      - name: Install search build tools
        run: npm ci --no-audit --no-fund

      - name: Cache embedding model
        uses: actions/cache@v4
        with:
          path: .cache/hf
          key: hf-bge-small-en-v1.5-q8-v1

      - name: Stage site and build search indexes
        run: |
          mkdir -p _site
          tar --exclude=./.git --exclude=./.github --exclude=./node_modules --exclude=./.cache --exclude=./_site -cf - . | tar -xf - -C _site
          node scripts/build-search.mjs _site _site
          du -sh _site/pagefind _site/search

      - name: Upload artifact
        uses: actions/upload-pages-artifact@v3
        with:
          path: '_site'

      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v4
